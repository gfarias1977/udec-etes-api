create or replace procedure p041_upd_campos_brecha_compra_detalles(IN p_proceso bigint, IN p_tipostk bigint, IN p_crea_log boolean)
    language plpgsql
as
$$
DECLARE
    v_message VARCHAR(500);
BEGIN
    IF p_crea_log THEN
        v_message := CONCAT('RUN [', public.fncgetdate()::TEXT,
                            '] Actualización campos Brecha compra detalle.');
        CALL public.add_process_log(p_proceso, v_message);
    END IF;

    UPDATE public.tbl_gaps_purchase_detail SET
        gappd_org_code       = public.fnc_instituciones_sede(gappd_camp_code),
        gappd_list_number    = 0,
        gappd_request_number = 0,
        gappd_description    = (
            SELECT SUBSTRING(
                       'T:' || COALESCE(item_value_01,'') ||
                       '. A:' || COALESCE(item_value_02,'') ||
                       '. E:' || COALESCE(item_value_03,''),
                       1, 255)
            FROM public.tbl_items
            WHERE item_code = gappd_item_code
        ),
        gappd_unit_value = COALESCE((
            SELECT item_unit_value
            FROM public.tbl_items
            WHERE item_code = gappd_item_code
        ), 0),
        gappd_cacc_code = (
            SELECT scho_cacc_code
            FROM public.tbl_schools
            WHERE scho_org_code = gappd_org_code
              AND scho_code     = gappd_scho_code
        )
    WHERE gappd_proc_id   = p_proceso
      AND gappd_stock_type = p_tipostk;

    RAISE NOTICE 'ACTUALIZA VALORES EN BRECHA COMPRA DETALLE OK';

    IF p_crea_log THEN
        v_message := CONCAT('RUN [', public.fncgetdate()::TEXT,
                            '] Actualización campos de conciliación Brecha compra detalle.');
        CALL public.add_process_log(p_proceso, v_message);
    END IF;
END;
$$;

alter procedure p041_upd_campos_brecha_compra_detalles(bigint, bigint, boolean) owner to neondb_owner;
