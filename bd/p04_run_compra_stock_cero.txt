create or replace procedure p04_run_compra_stock_cero(IN p_proceso bigint)
    language plpgsql
as
$$
DECLARE
    v_tipostk  FLOAT   := 0;
    v_o_result INTEGER := 0;
    v_message  VARCHAR(500);
    v_row_count BIGINT;
BEGIN
    DELETE FROM public.tbl_gaps_purchase        WHERE gapp_proc_id  = p_proceso;
    DELETE FROM public.tbl_gaps_purchase_detail WHERE gappd_proc_id = p_proceso;

    v_message := CONCAT('COMPRA-STOCK-CERO: RUN [', NOW()::TEXT, '] Inserción Brecha compra.');
    CALL public.add_process_log(p_proceso, v_message);

    INSERT INTO public.tbl_gaps_purchase (
        gapp_proc_id, gapp_stock_type, gapp_year, gapp_city_code,
        gapp_item_code, gapp_initial_gap, gapp_optimized_gap, gapp_item_status, gapp_unit_value
    )
    SELECT DISTINCT
        gapr_proc_id, v_tipostk, gapr_academic_year, gapr_city_code, gapr_item_code,
        (gapr_initial_gap * -1), (gapr_optimized_gap * -1), gapr_item_active, 0
    FROM (
        SELECT DISTINCT gapr_proc_id, gapr_academic_year, gapr_city_code, gapr_item_code,
               CASE gapr_item_active WHEN 'S' THEN gapr_gap ELSE 0 END gapr_initial_gap,
               CASE gapr_item_active WHEN 'S' THEN gapr_gap ELSE 0 END gapr_optimized_gap,
               gapr_demand, gapr_item_active
        FROM public.tbl_gaps_dda_vs_stock
        WHERE gapr_proc_id=p_proceso AND gapr_national_stock=0 AND gapr_demand>0
          AND gapr_org_code='*' AND gapr_scho_code='*' AND gapr_cours_code='*'
          AND gapr_camp_code='*' AND gapr_city_stock_di=0
        UNION ALL
        SELECT DISTINCT gapr_proc_id, gapr_academic_year, gapr_city_code, gapr_item_code,
               CASE gapr_item_active WHEN 'S' THEN gapr_gap ELSE 0 END,
               CASE gapr_item_active WHEN 'S' THEN gapr_gap ELSE 0 END,
               gapr_demand, gapr_item_active
        FROM public.tbl_gaps_stock_vs_dda
        WHERE gapr_proc_id=p_proceso AND gapr_national_stock=0 AND gapr_demand>0
          AND gapr_org_code='*' AND gapr_scho_code='*' AND gapr_cours_code='*'
          AND gapr_camp_code='*' AND gapr_city_stock_di=0
    ) s
    ON CONFLICT (gapp_proc_id, gapp_stock_type, gapp_year, gapp_city_code, gapp_item_code) DO NOTHING;

    CALL public.p041_upd_campos_brecha_compra(p_proceso, v_tipostk::BIGINT, TRUE);

    v_message := CONCAT('COMPRA-STOCK-CERO: RUN [', NOW()::TEXT, '] Inserción Brecha compra detalle.');
    CALL public.add_process_log(p_proceso, v_message);

    INSERT INTO public.tbl_gaps_purchase_detail (
        gappd_proc_id, gappd_stock_type, gappd_year, gappd_city_code, gappd_org_code,
        gappd_scho_code, gappd_cours_code, gappd_item_code, gappd_camp_code,
        gappd_initial_gap, gappd_optimized_gap, gappd_item_status, gappd_unit_value,
        gappd_list_number, gappd_request_number
    )
    SELECT DISTINCT
        gapr_proc_id, v_tipostk, gapr_academic_year, gapr_city_code, gapr_org_code,
        gapr_scho_code, gapr_cours_code, gapr_item_code, gapr_camp_code,
        (gapr_initial_gap * -1), (gapr_optimized_gap * -1), gapr_item_active, 0,
        0, 0
    FROM (
        SELECT DISTINCT gapr_proc_id, gapr_academic_year, gapr_city_code, gapr_org_code,
               gapr_scho_code, gapr_cours_code, gapr_item_code, gapr_camp_code,
               CASE gapr_item_active WHEN 'S' THEN gapr_gap ELSE 0 END gapr_initial_gap,
               CASE gapr_item_active WHEN 'S' THEN gapr_gap ELSE 0 END gapr_optimized_gap,
               gapr_item_active
        FROM public.tbl_gaps_dda_vs_stock
        WHERE gapr_proc_id=p_proceso AND gapr_national_stock=0 AND gapr_demand>0
          AND gapr_org_code<>'*' AND gapr_scho_code<>'*' AND gapr_cours_code<>'*'
          AND gapr_camp_code<>'*' AND gapr_city_stock_di=0
        UNION ALL
        SELECT DISTINCT gapr_proc_id, gapr_academic_year, gapr_city_code, gapr_org_code,
               gapr_scho_code, gapr_cours_code, gapr_item_code, gapr_camp_code,
               CASE gapr_item_active WHEN 'S' THEN gapr_gap ELSE 0 END,
               CASE gapr_item_active WHEN 'S' THEN gapr_gap ELSE 0 END,
               gapr_item_active
        FROM public.tbl_gaps_stock_vs_dda
        WHERE gapr_proc_id=p_proceso AND gapr_national_stock=0 AND gapr_demand>0
          AND gapr_org_code<>'*' AND gapr_scho_code<>'*' AND gapr_cours_code<>'*'
          AND gapr_camp_code<>'*' AND gapr_city_stock_di=0
    ) s;

    GET DIAGNOSTICS v_row_count = ROW_COUNT;
    v_o_result := v_row_count;

    CALL public.p041_upd_campos_brecha_compra_detalles(p_proceso, v_tipostk::BIGINT, TRUE);

    RAISE NOTICE 'ACTUALIZA VALORES BRECHA FINAL EN DETALLE OK. Registros: %', v_o_result;

EXCEPTION WHEN OTHERS THEN
    v_message := CONCAT('COMPRA-STOCK-CERO: ERROR: ', SQLSTATE, ' - ', SQLERRM);
    CALL public.add_process_log(p_proceso, v_message);
    RAISE;
END;
$$;

alter procedure p04_run_compra_stock_cero(bigint) owner to neondb_owner;
